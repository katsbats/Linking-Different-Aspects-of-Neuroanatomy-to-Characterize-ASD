#### MNI brain masks ####
# Resample the FSL MNI152 brain mask to the resolution of the data.
# -rmode NN = nearest-neighbour interpolation, so the mask stays binary (0/1)

# 1.5 mm mask: used for the coverage check (1.coveragecheck90.sh)
# uses AFNI 
~/abin/3dresample \
  -master '/Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Input/ABIDE_I/Caltech_51465/mri/mwp1mprage.nii' \
  -input '/Users/katharinabremer/fsl/data/standard/MNI152_T1_1mm_brain_mask.nii.gz' \
  -prefix '/Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI_mask_resampled1p5.nii' \
  -rmode NN

# 4 mm mask: used for the preprocessing
~/abin/3dresample \
  -master '/Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI152_T1_4mm_brain.nii.gz' \
  -input '/Users/katharinabremer/fsl/data/standard/MNI152_T1_2mm_brain_mask.nii.gz' \
  -prefix '/Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI_mask_resampled4mm.nii' \
  -rmode NN

# Sanity checks: voxel size should be 4 mm, value range should be 0 to 1 (binary)
fslinfo /Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI_mask_resampled4mm.nii | grep pixdim
fslstats /Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI_mask_resampled4mm.nii -R

# Compress mask
gzip /Volumes/g_psyplafor_methlab$/Students/Katharina_Bremer/Data/MNImask/MNI_mask_resampled4mm.nii
